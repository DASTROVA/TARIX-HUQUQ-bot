# =====================================
# FOYDALANUVCHINI RO‘YXATGA OLISH
# =====================================

def register_user(user):
    user_id = user.id

    if user_id not in users:
        users[user_id] = {
            "name": user.full_name,
            "tests": 0,
            "correct": 0,
            "points": 0
        }


# =====================================
# TESTNI BOSHLASH
# =====================================

async def start_test(message, test_type):
    register_user(message.from_user)

    if test_type == "tarix":
        questions = TARIX_QUESTIONS
        title = "📚 TARIX TESTI"
    elif test_type == "huquq":
        questions = HUQUQ_QUESTIONS
        title = "⚖️ HUQUQ TESTI"
    else:
        questions = TARIX_QUESTIONS + HUQUQ_QUESTIONS
        title = "📝 UMUMIY IMTIHON"

    user_id = message.from_user.id

    active_tests[user_id] = {
        "questions": questions,
        "current": 0,
        "correct": 0
    }

    await message.answer(
        f"{title}\n\nTest boshlandi!",
        reply_markup=InlineKeyboardMarkup(
            inline_keyboard=[
                [
                    InlineKeyboardButton(
                        text="▶️ Boshlash",
                        callback_data="next_question"
                    )
                ]
            ]
        )
    )


# =====================================
# SAVOLNI CHIQARISH
# =====================================

async def show_question(callback):
    user_id = callback.from_user.id

    if user_id not in active_tests:
        await callback.answer("Avval testni boshlang!")
        return

    test = active_tests[user_id]
    index = test["current"]
    questions = test["questions"]

    if index >= len(questions):
        await finish_test(callback)
        return

    question = questions[index]

    buttons = []

    for i, answer in enumerate(question["answers"]):
        buttons.append([
            InlineKeyboardButton(
                text=answer,
                callback_data=f"answer_{i}"
            )
        ])

    await callback.message.edit_text(
        f"❓ SAVOL {index + 1}/{len(questions)}\n\n"
        f"{question['question']}",
        reply_markup=InlineKeyboardMarkup(
            inline_keyboard=buttons
        )
    )


# =====================================
# JAVOBNI TEKSHIRISH
# =====================================

@dp.callback_query(F.data.startswith("answer_"))
async def check_answer(callback: CallbackQuery):
    user_id = callback.from_user.id

    if user_id not in active_tests:
        await callback.answer("Test faol emas!")
        return

    test = active_tests[user_id]
    index = test["current"]

    if index >= len(test["questions"]):
        await callback.answer("Test tugagan!")
        return

    question = test["questions"][index]
    selected = int(callback.data.split("_")[1])

    if selected == question["correct"]:
        test["correct"] += 1
        result = "✅ To‘g‘ri javob!"
        users[user_id]["correct"] += 1
        users[user_id]["points"] += 10
    else:
        correct_answer = question["answers"][question["correct"]]
        result = f"❌ Noto‘g‘ri javob!\nTo‘g‘ri javob: {correct_answer}"

    explanation = question["explanation"]

    test["current"] += 1

    await callback.message.edit_text(
        f"{result}\n\n"
        f"📖 Izoh: {explanation}",
        reply_markup=InlineKeyboardMarkup(
            inline_keyboard=[
                [
                    InlineKeyboardButton(
                        text="➡️ Keyingi savol",
                        callback_data="next_question"
                    )
                ]
            ]
        )
    )

    await callback.answer()


# =====================================
# KEYINGI SAVOL
# =====================================

@dp.callback_query(F.data == "next_question")
async def next_question(callback: CallbackQuery):
    await show_question(callback)
    await callback.answer()


# =====================================
# TESTNI YAKUNLASH
# =====================================

async def finish_test(callback):
    user_id = callback.from_user.id
    test = active_tests[user_id]

    total = len(test["questions"])
    correct = test["correct"]
    wrong = total - correct

    users[user_id]["tests"] += 1

    await callback.message.edit_text(
        "🏁 TEST YAKUNLANDI!\n\n"
        f"📚 Jami savollar: {total}\n"
        f"✅ To‘g‘ri javoblar: {correct}\n"
        f"❌ Noto‘g‘ri javoblar: {wrong}\n"
        f"🏆 Ball: {correct * 10}\n\n"
        "Yana test ishlash uchun /start bosing."
    )

    del active_tests[user_id]

@dp.message(Command("rating"))
async def rating_command(message: Message):
    if not users:
        await message.answer("Hozircha reyting bo‘sh.")
        return

    ranking = sorted(
        users.items(),
        key=lambda item: item[1]["points"],
        reverse=True
    )

    text = "🏆 FOYDALANUVCHILAR REYTINGI\n\n"

    for i, (user_id, data) in enumerate(ranking[:10], start=1):
        text += (
            f"{i}. {data['name']} — "
            f"{data['points']} ball\n"
        )

    await message.answer(text)

# =====================================

async def main():
    if not TOKEN:
        raise RuntimeError(
            "BOT_TOKEN sozlanmagan! Railway Variables bo‘limiga token kiriting."
        )

    bot = Bot(token=TOKEN)

    print("Bot ishga tushdi!")

    try:
        await dp.start_polling(bot)
    finally:
        await bot.session.close()


if __name__ == "__main__":
    asyncio.run(main())
